35class _NoWriteProgressStorage(kopf.ProgressStorage): 

36 """Progress storage that keeps state in memory and writes nothing to Kubernetes. 

37 

38 Kopf's default storage writes annotations and finalizers to watched objects, 

39 requiring namespaces/patch permissions we deliberately do not grant. Namespace 

40 handlers are idempotent so losing state on restart is acceptable. 

41 

42 Methods must be plain (non-async) — Kopf calls them synchronously. 

43 """

46 return None

57 def clear(self, *, essence): 

58 # clear() is a transform, not a void method: kopf feeds both sides of 

59 # its diff through it (processing.py) and expects the essence back with 

60 # any progress annotations stripped. Returning None nulled `new` as well 

61 # as `old`, which meant diffbase_storage.store() was never reached — 

62 # it is guarded by `if cause.new is not None`. That defeated diff-base 

63 # tracking entirely, whatever the diff-base storage did. See #65. 

64 # 

65 # We write no progress annotations, so there is nothing to strip; defer 

66 # to the base implementation, which deep-copies so callers cannot mutate 

67 # a stored essence. 

68 return super().clear(essence=essence)

71class _InMemoryDiffBaseStorage(kopf.DiffBaseStorage): 

72 """Diff-base storage kept in this process rather than on the object. 

73 

74 Kopf's default AnnotationsDiffBaseStorage records the last-handled essence 

75 as an annotation on every watched object, which needs the namespace patch 

76 permission this operator deliberately does not hold. 

77 

78 Returning None unconditionally — the previous behaviour — is not a neutral 

79 stand-in for that. Kopf reads "no stored essence" as `Reason.CREATE` 

80 (causes.py: `if old is None`), so *every* namespace event re-ran the create 

81 handler: a label edit, an annotation change, the initial listing. Storing 

82 the essence in memory restores real CREATE/UPDATE/NOOP detection without 

83 needing to write to the object. 

84 

85 Losing this on restart is deliberate and safe. Bootstrap and reconcile 

86 converge every namespace anyway, and the create handler is idempotent and 

87 rollout-serialised. Note that it also means RESUME stays unreachable: a 

88 fresh process has no essence for pre-existing namespaces, so they are still 

89 classified as CREATE on the first listing after a restart. 

90 

91 Methods must be plain (non-async) — Kopf calls them synchronously. 

92 """

94 def __init__(self) -> None: 

95 super().__init__() 

96 # Plain dict: get/set/pop are atomic under the GIL, which is enough for 

97 # the concurrent handler invocations kopf makes. 

98 self._essences: dict[str, object] = {} 

99 

100 @staticmethod 

101 def _uid(body) -> str | None: 

102 return (body.get("metadata") or {}).get("uid") 

103 

104 def fetch(self, *, body): 

105 uid = self._uid(body) 

106 return self._essences.get(uid) if uid else None 

107 

108 def store(self, *, body, patch, essence): 

109 uid = self._uid(body) 

110 if uid: 

111 self._essences[uid] = essence 

112 

113 def retain(self, live_uids) -> int: 

114 """Drop essences for objects that no longer exist, returning the count. 

115 

116 Kopf offers no eviction hook on this interface, so reconcile prunes 

117 against the namespace list it already fetches. Without this the dict 

118 grows for the operator's whole lifetime on clusters that churn 

119 namespaces, such as per-build CI namespaces. 

120 """ 

121 stale = self._essences.keys() - set(live_uids) 

122 for uid in stale: 

123 self._essences.pop(uid, None) 

124 return len(stale) 

125 

126 def __len__(self) -> int: 

127 return len(self._essences)

148def _error_class(exc: Exception) -> str: 

149 """Whether a failure came from the API server or from our own code. 

150 

151 An ApiException is an operational condition — a 403 from an admission 

152 webhook, a 409, a 503 — and retrying next pass is the right response. Any 

153 other exception reaching a reconcile handler is a defect in this operator 

154 and wants a human. Both stay non-fatal for the remaining namespaces; the 

155 label is what tells them apart. 

156 """ 

157 return "api" if isinstance(exc, client.ApiException) else "internal"

193 """State of the Helm-owned central checker Deployment: ready, unavailable or missing. 

194

217def _central_namespace_status( 

218 audited: bool, labels: dict, policy: dict, central_state: str, grant_error: str | None, 

219) -> dict: 

220 """Per-namespace IAP status when one cluster-wide checker covers every namespace. 

221 

222 `deployed` keeps meaning "this namespace is being audited", so here it 

223 tracks the single central Deployment rather than a checker of the 

224 namespace's own. Reporting it per namespace rather than once keeps the 

225 status field's shape identical across modes, so anything reading it does 

226 not need to know which mode the cluster is in. 

227 """ 

228 if not audited: 

229 return {"deployed": False, "reason": _not_audited_reason(labels, policy)} 

230 if grant_error: 

231 # The checker is running but cannot read this namespace's pull secrets, 

232 # so private images here will read as unavailable. Not "deployed". 

233 return {"deployed": False, "reason": f"secret grant failed: {grant_error}"} 

234 if central_state == "missing": 

235 return {"deployed": False, "reason": "central checker Deployment not found"} 

236 if central_state != "ready": 

237 return {"deployed": False, "reason": "central checker unavailable"} 

238 return {"deployed": True, "reason": "audited by central checker"}

241def _bootstrap_checkers() -> None: 

242 from kubeic_operator.deployer import ( 

243 CENTRAL_MODE, CHECKER_ENABLED, deploy_checker_serialised, get_secret_names_for_namespace, 

244 ) 

245 from kubeic_operator.handlers.namespace import _should_deploy_checker, _get_effective_policy 

246 

247 if not CHECKER_ENABLED: 

248 logger.info("Checkers disabled (checker.enabled=false); skipping bootstrap") 

249 return 

250 

251 if CENTRAL_MODE: 

252 # Nothing to pace: there are no per-namespace checker pods to create. 

253 # Reconcile, which runs immediately after this, converges the central 

254 # checker's per-namespace secret grants and drains any checkers left 

255 # over from perNamespace mode. 

256 logger.info("Central mode (checker.mode=central); skipping per-namespace bootstrap") 

257 return 

258 

259 v1 = client.CoreV1Api() 

260 try: 

261 namespaces = v1.list_namespace().items 

262 except client.ApiException as exc: 

263 logger.error("Failed to list namespaces during bootstrap: %s", exc) 

264 return 

265 

266 for ns in namespaces: 

267 name = ns.metadata.name 

268 labels = ns.metadata.labels or {} 

269 policy = _get_effective_policy(name) 

270 if not _should_deploy_checker(name, labels, policy): 

271 continue 

272 interval = policy.get("availability", {}).get("intervalMinutes", 30) 

273 cred_source = policy.get("credentialSource", {}).get("type", "pullSecret") 

274 try: 

275 # blocking=True: wait our turn and then for the rollout, so the 

276 # whole bootstrap is one checker at a time. 

277 deploy_checker_serialised(name, blocking=True, 

278 check_interval_minutes=interval, credential_source=cred_source, 

279 secret_names=get_secret_names_for_namespace(name)) 

280 logger.info("Bootstrapped checker in namespace %s", name) 

281 except Exception as exc: 

282 _record_failure(name, "deploy", exc)

285def _reconcile_checkers() -> dict: 

286 """Ensure checker state matches desired state for all namespaces. 

287 

288 Returns a dict of namespace -> {deployed, reason} for status reporting. 

289 """ 

290 from kubeic_operator.deployer import ( 

291 CENTRAL_MODE, CHECKER_DEPLOYMENT, CHECKER_ENABLED, deploy_checker_serialised, 

292 ensure_central_secret_access, teardown_central_secret_access, 

293 teardown_checker_serialised, 

294 get_secret_names_for_namespace, 

295 ) 

296 from kubeic_operator.handlers.namespace import ( 

297 _should_audit, _should_deploy_checker, _get_effective_policy, 

298 ) 

299 

300 v1 = client.CoreV1Api() 

301 apps_v1 = client.AppsV1Api() 

302 try: 

303 namespaces = v1.list_namespace().items 

304 except client.ApiException as exc: 

305 logger.error("Failed to list namespaces during reconcile: %s", exc) 

306 return {} 

307 

308 # Kopf has no eviction hook on the diff-base interface, so prune here 

309 # against the namespace list this pass already fetched. 

310 evicted = DIFFBASE_STORAGE.retain({ns.metadata.uid for ns in namespaces}) 

311 if evicted: 

312 logger.debug("Evicted %d diff-base entries for deleted namespaces", evicted) 

313 

314 # One probe per pass, not per namespace: in central mode every namespace's 

315 # coverage comes from this single Deployment. 

316 central_state = _central_checker_state(apps_v1) if CENTRAL_MODE and CHECKER_ENABLED else "missing" 

317 

318 namespace_status = {} 

319 for ns in namespaces: 

320 name = ns.metadata.name 

321 labels = ns.metadata.labels or {} 

322 policy = _get_effective_policy(name) 

323 audited = _should_audit(name, labels, policy) 

324 should = _should_deploy_checker(name, labels, policy) 

325 

326 # Central mode binds the one checker into each audited namespace so it 

327 # can read that namespace's pull secrets, and unbinds it everywhere 

328 # else. Done before the pod bookkeeping below because the two are 

329 # independent: during a switch from perNamespace the grants need to 

330 # exist while the old checkers are still draining. 

331 grant_error = None 

332 if CENTRAL_MODE: 

333 try: 

334 if audited: 

335 ensure_central_secret_access(name, get_secret_names_for_namespace(name)) 

336 elif teardown_central_secret_access(name): 

337 logger.info("Reconciled: revoked central checker secret access to %s", name) 

338 except Exception as exc: 

339 _record_failure(name, "secret-grant", exc) 

340 grant_error = _failure_reason(exc) 

341 

342 checker_exists = False 

343 try: 

344 apps_v1.read_namespaced_deployment(CHECKER_DEPLOYMENT, name) 

345 checker_exists = True 

346 except client.ApiException as exc: 

347 if exc.status != 404: 

348 # Previously this re-raised, which escaped the loop entirely: 

349 # one namespace answering 403 or 503 left every namespace after 

350 # it unreconciled, and aborted the pass before the status was 

351 # written, so the staleness was silent. Skip this namespace and 

352 # let the next pass retry it. 

353 _record_failure(name, "probe", exc) 

354 # In central mode this probe only looks for a leftover 

355 # perNamespace checker, so failing it says nothing about whether 

356 # the namespace is being audited — the central checker answers 

357 # that. The failure is still counted above. 

358 namespace_status[name] = ( 

359 _central_namespace_status(audited, labels, policy, central_state, grant_error) 

360 if CENTRAL_MODE 

361 else {"deployed": False, "reason": f"state unknown: {_failure_reason(exc)}"} 

362 ) 

363 continue 

364 

365 if should and not checker_exists: 

366 interval = policy.get("availability", {}).get("intervalMinutes", 30) 

367 cred_source = policy.get("credentialSource", {}).get("type", "pullSecret") 

368 try: 

369 # Paced for the same reason as bootstrap: on a fresh cluster 

370 # reconcile is the one creating every checker. 

371 deploy_checker_serialised(name, blocking=True, 

372 check_interval_minutes=interval, 

373 credential_source=cred_source, 

374 secret_names=get_secret_names_for_namespace(name)) 

375 logger.info("Reconciled: deployed checker in %s", name) 

376 namespace_status[name] = {"deployed": True} 

377 except Exception as exc: 

378 # Status is written inside the branches so it records the 

379 # outcome rather than the intent. Reporting deployed: true here 

380 # made a namespace that failed every pass indistinguishable 

381 # from a healthy one. 

382 _record_failure(name, "deploy", exc) 

383 namespace_status[name] = { 

384 "deployed": False, 

385 "reason": f"deploy failed: {_failure_reason(exc)}", 

386 } 

387 elif not should and checker_exists: 

388 try: 

389 # Serialised so a mass teardown (checker.enabled: false turns 

390 # every namespace off at once) cannot interleave with a deploy 

391 # still in flight from the namespace handler. 

392 teardown_checker_serialised(name, blocking=True) 

393 logger.info("Reconciled: removed checker from %s", name) 

394 except Exception as exc: 

395 _record_failure(name, "teardown", exc) 

396 # The checker is still there, so deployed stays true. 

397 namespace_status[name] = { 

398 "deployed": True, 

399 "reason": f"teardown failed: {_failure_reason(exc)}", 

400 } 

401 continue 

402 namespace_status[name] = { 

403 "deployed": False, 

404 "reason": "central mode" if CENTRAL_MODE else _not_audited_reason(labels, policy), 

405 } 

406 elif should: 

407 namespace_status[name] = {"deployed": True} 

408 

409 # In central mode the branches above only ever drain checkers left from 

410 # perNamespace mode, so whatever they concluded describes a pod that is 

411 # meant to be gone. The namespace's real coverage is the central 

412 # checker, so that is what the status reports. 

413 if CENTRAL_MODE: 

414 namespace_status[name] = _central_namespace_status( 

415 audited, labels, policy, central_state, grant_error, 

416 ) 

417 

418 return namespace_status

441def _run_cluster_audit() -> None: 

442 v1 = client.CoreV1Api() 

443 try: 

444 pods = v1.list_pod_for_all_namespaces() 

445 except client.ApiException as exc: 

446 logger.error("Failed to list pods: %s", exc) 

447 return 

448 

449 pod_list = [] 

450 for pod in pods.items: 

451 # Terminated pods (Succeeded/Failed) never pull again and lingering 

452 # ones (e.g. dead CI job pods) skew prerelease/spread with images 

453 # nothing is running. 

454 if pod.status.phase in ("Succeeded", "Failed"): 

455 continue 

456 pod_list.append({ 

457 "metadata": { 

458 "name": pod.metadata.name, 

459 "namespace": pod.metadata.namespace, 

460 "creationTimestamp": pod.metadata.creation_timestamp.isoformat() if pod.metadata.creation_timestamp else "", 

461 "annotations": pod.metadata.annotations or {}, 

462 }, 

463 "status": { 

464 "startTime": pod.status.start_time.isoformat() if pod.status.start_time else "", 

465 }, 

466 "spec": { 

467 "containers": [{"name": c.name, "image": c.image} for c in (pod.spec.containers or [])], 

468 "initContainers": [{"name": c.name, "image": c.image} for c in (pod.spec.init_containers or [])], 

469 }, 

470 }) 

471 

472 policy = _get_default_policy() 

473 max_age_days = policy.get("prerelease", {}).get("maxAgeDays", 7) 

474 stable_suffixes = policy.get("prerelease", {}).get("stableSuffixes") 

475 skip_annotation = policy.get("skipAnnotation") or None 

476 spread_threshold = policy.get("versionSpread", {}).get("threshold", 3) 

477 

478 prerelease_findings = check_prerelease(pod_list, max_age_days=max_age_days, stable_suffixes=stable_suffixes, skip_annotation=skip_annotation) 

479 violations = filter_violations(prerelease_findings, max_age_days=max_age_days) 

480 update_prerelease_metrics(prerelease_findings, violations=violations) 

481 if violations: 

482 logger.warning("Found %d pre-release violations (max_age=%dd)", len(violations), max_age_days) 

483 

484 spread_findings = aggregate_version_spread(pod_list, threshold=spread_threshold, skip_annotation=skip_annotation) 

485 update_spread_metrics(spread_findings) 

486 spread_violations = [f for f in spread_findings if f.violates_threshold] 

487 if spread_violations: 

488 logger.warning("Found %d version spread violations (threshold=%d)", len(spread_violations), spread_threshold) 

489 

490 logger.info("Cluster audit complete: %d pre-release findings, %d spread findings", 

491 len(prerelease_findings), len(spread_findings))

494def _audit_loop() -> None: 

495 # Publish the cluster-wide metrics first. The pre-release and version-spread 

496 # checks need nothing from the checkers, and the paced rollout below can take 

497 # many minutes on a large cluster — running it first would leave the operator 

498 # exporting no metrics for that whole window. 

499 try: 

500 _run_cluster_audit() 

501 except Exception: 

502 logger.exception("Initial cluster audit failed") 

503 

504 # Bootstrap runs here rather than in on_startup because it is now paced: 

505 # readiness-gating 50+ namespaces takes minutes, and kopf does not begin 

506 # watching anything until the startup handler returns. Blocking there would 

507 # leave the operator blind to namespace events for the whole rollout. 

508 try: 

509 _bootstrap_checkers() 

510 except Exception: 

511 logger.exception("Checker bootstrap failed") 

512 try: 

513 namespace_status = _reconcile_checkers() 

514 if namespace_status: 

515 _write_iap_status(namespace_status) 

516 except Exception: 

517 logger.exception("Initial checker reconciliation failed") 

518 

519 while True: 

520 time.sleep(SCAN_INTERVAL) 

521 try: 

522 _run_cluster_audit() 

523 except Exception: 

524 logger.exception("Cluster audit failed") 

525 try: 

526 namespace_status = _reconcile_checkers() 

527 if namespace_status: 

528 _write_iap_status(namespace_status) 

529 except Exception: 

530 logger.exception("Checker reconciliation failed")

533@kopf.on.startup() 

534def on_startup(settings: kopf.OperatorSettings, **kwargs): 

535 # Kopf defaults to writing progress state as annotations on watched objects 

536 # and adding finalizers to track in-flight handlers. Both require namespace 

537 # patch permissions we intentionally don't hold. In-memory storage is fine 

538 # for namespace handlers — they complete quickly and are idempotent on retry. 

539 settings.persistence.progress_storage = _NoWriteProgressStorage() 

540 settings.persistence.diffbase_storage = DIFFBASE_STORAGE 

541 settings.persistence.finalizer = None 

542 

543 start_http_server(METRICS_PORT) 

544 logger.info("Prometheus metrics server started on port %d", METRICS_PORT) 

545 try: 

546 k8s_config.load_incluster_config() 

547 except k8s_config.ConfigException: 

548 k8s_config.load_kube_config() 

549 # The audit thread owns bootstrap and reconcile. Returning promptly is what 

550 # lets kopf start watching namespaces while the paced rollout is still 

551 # working through the cluster. 

552 threading.Thread(target=_audit_loop, daemon=True, name="audit-loop").start()
